const ADMIN_PIN = "1234";

let isAdmin = false;

let products = JSON.parse(localStorage.getItem("products")) || [
{name:"콜라",icon:"🥤",stock:10},
{name:"사이다",icon:"🍋",stock:8},
{name:"캔커피",icon:"☕",stock:5},
{name:"물",icon:"💧",stock:12},
{name:"핫식스",icon:"⚡",stock:3},
{name:"몬스터",icon:"👹",stock:2}
];

// 저장
function saveData(){
localStorage.setItem("products", JSON.stringify(products));
localStorage.setItem("lastUpdate", new Date().toLocaleString());
updateStats();
}

// 통계
function updateStats(){

```
document.getElementById("productCount").textContent =
    products.length;

document.getElementById("totalStock").textContent =
    products.reduce((sum,p)=>sum+p.stock,0);

document.getElementById("soldOutCount").textContent =
    products.filter(p=>p.stock===0).length;

document.getElementById("lastUpdate").textContent =
    localStorage.getItem("lastUpdate") || "-";
```

}

// 상품 출력
function renderProducts(){

```
const keyword =
    document.getElementById("search")
    .value
    .toLowerCase();

const container =
    document.getElementById("products");

container.innerHTML = "";

products.forEach((p,index)=>{

    if(!p.name.toLowerCase().includes(keyword))
        return;

    const low = p.stock <= 3;
    const soldout = p.stock === 0;

    container.innerHTML += `
    <div class="card">

        <div class="product-header">

            <div class="icon">${p.icon}</div>

            <div>
                <h3>${p.name}</h3>

                <div class="stock ${low ? "low" : ""}">
                    재고 ${p.stock}개
                </div>

                ${
                    soldout
                    ? `<div class="soldout">품절</div>`
                    : ""
                }

                ${
                    low && !soldout
                    ? `<div style="margin-top:5px;color:#f59e0b;font-weight:bold">
                        재고 부족
                       </div>`
                    : ""
                }

            </div>
        </div>

        <div class="controls ${!isAdmin ? "locked" : ""}">

            <button onclick="changeStock(${index},1)">
                +
            </button>

            <button onclick="changeStock(${index},-1)">
                -
            </button>

            <input
                type="number"
                min="0"
                value="${p.stock}"
                onchange="setStock(${index},this.value)"
            >

        </div>

    </div>
    `;
});

updateStats();
```

}

// 재고 변경
function changeStock(index,val){

```
if(!isAdmin) return;

products[index].stock += val;

if(products[index].stock < 0){
    products[index].stock = 0;
}

saveData();
renderProducts();
```

}

// 직접 입력
function setStock(index,val){

```
if(!isAdmin) return;

products[index].stock =
    Math.max(0, parseInt(val) || 0);

saveData();
renderProducts();
```

}

// 관리자
function adminLogin(){

```
if(isAdmin){

    isAdmin = false;

    alert("관리자 모드 종료");

    renderProducts();

    return;
}

const pin = prompt("관리자 PIN 입력");

if(pin === ADMIN_PIN){

    isAdmin = true;

    alert("관리자 모드 활성화");

    renderProducts();

}else{

    alert("PIN 오류");
}
```

}

// 다크모드
function toggleTheme(){

```
document.body.classList.toggle("dark");

localStorage.setItem(
    "theme",
    document.body.classList.contains("dark")
    ? "dark"
    : "light"
);
```

}

// 초기화
function resetData(){

```
if(
    !confirm(
        "모든 상품 재고를 0으로 초기화할까요?"
    )
){
    return;
}

products.forEach(p=>{
    p.stock = 0;
});

saveData();
renderProducts();
```

}

// CSV 내보내기
function exportCSV(){

```
let csv = "상품명,재고\n";

products.forEach(p=>{

    csv += `${p.name},${p.stock}\n`;

});

const blob =
    new Blob(
        [csv],
        {type:"text/csv;charset=utf-8;"}
    );

const a =
    document.createElement("a");

a.href =
    URL.createObjectURL(blob);

a.download =
    "vending_stock.csv";

a.click();
```

}

// CSV 가져오기
function importCSV(event){

```
const file =
    event.target.files[0];

if(!file) return;

const reader =
    new FileReader();

reader.onload = function(e){

    const rows =
        e.target.result
        .trim()
        .split("\n");

    rows.shift();

    rows.forEach(row=>{

        const [name,stock] =
            row.split(",");

        if(!name) return;

        const item =
            products.find(
                p=>p.name.trim()===name.trim()
            );

        if(item){

            item.stock =
                parseInt(stock) || 0;

        }else{

            products.push({
                name:name.trim(),
                icon:"📦",
                stock:parseInt(stock) || 0
            });
        }
    });

    saveData();
    renderProducts();

    alert("CSV 가져오기 완료");
};

reader.readAsText(file);
```

}

// 다크모드 복원
if(localStorage.getItem("theme")==="dark"){
document.body.classList.add("dark");
}

// 시작
renderProducts();
